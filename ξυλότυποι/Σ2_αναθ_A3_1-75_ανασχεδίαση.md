# Σ2 αναθ. 24/09/2026 – Α3 1:75 «όσο πιο μεγάλο γίνεται» (26/10/2026)

Παραδοτέο: `Σ2_αναθ_24-09-26_Ξυλότυπος_οροφής_υπογείου_A3_1-75.pdf` (+PNG 200 dpi). Α3 οριζόντιο 420×297, κλίμακα 1:75 (1 m = 13.33 mm).

- Γιατί 1:75: είναι η μεγαλύτερη τυποποιημένη κλίμακα (υπάρχει στο τριγωνικό κλιμακόμετρο) που χωράει σε Α3 με τις αλυσίδες διαστάσεων· το 1:50 θέλει Α2, το 1:60 δεν χωράει. Περιθώρια 17.3 mm αριστερά/δεξιά, 15.6 mm πάνω/κάτω.
- Επαλήθευση κλίμακας στο PDF: συνολική διάσταση 22.10 → 294.69 mm (αναμενόμενο 294.67), γραφική κλίμακα 2 m → 26.67 mm.
- Πινακίδα: μόνο τίτλος (ξυλότυπος οροφής υπογείου ±0,00 / +0,30, αναθ. 24/09/2026), κλίμακα 1:75 (Α3) με γραφική κλίμακα, διεύθυνση.
- Μέθοδος (ο χώρος εργασίας είχε μηδενιστεί – δεν υπήρχαν πια τα πηγαία PDF): το φύλλο Α2 1:50 της αναθεώρησης ξαναστήθηκε σε Α3 με ακριβή συντελεστή 2/3 (`enlarge.py`): η γεωμετρία ξανασχεδιάζεται με τα αρχικά πάχη γραμμών, και κάθε κείμενο μεγαλώνει ×1.2 (ή λιγότερο, ανά κείμενο, μόνο όπου θα δημιουργούσε νέα σύγκρουση με κείμενο, τοίχο/δοκό/υποστύλωμα, γραμμή διάστασης, tick ή μπλε φυτευτό). Αγκύρωση ανά ρόλο: τιμές διαστάσεων κρατούν την πλευρά προς τη γραμμή τους, ετικέτες δοκών/κύκλοι πλακών το κέντρο τους, ετικέτες/στάθμες/σημειώσεις την αρχή της γραμμής βάσης, πολύγραμμα μπλοκ τις αναλογίες τους. Οι κύκλοι πλακών μεγαλώνουν 25 % παραπάνω ώστε το «h=..» να χωράει.
- Διορθώσεις έναντι του Α2: λευκή μάσκα πίσω από τις τιμές πλάτους υποστυλωμάτων (η γραμμή αναφοράς της σειράς αξόνων περνούσε μέσα από το κείμενο), Κ30 κάτω από τον νότιο τοίχο (όπως το Κ28), ετικέτα Κ25 (25/50) εκτός της αλυσίδας V3, ετικέτα Δ2.1 (3 γραμμές) ανάμεσα στο μπλε σύμβολο στάθμης και την αλυσίδα H1, τιμή 1.60 του πυρήνα εκτός της γραμμής 1.80, «Φελιζόλ» εκτός της γραμμής 2.15, Δ8.1 εκτός του tick της διάστασης 0.20. Τελικός έλεγχος: κανένα κείμενο πάνω σε γραμμή διάστασης/tick, καμία νέα επικάλυψη.
- Μεγέθη στο Α3: διαστάσεις 7.2 pt (εσωτερικές 6.6), ετικέτες υποστυλωμάτων 7.6 pt έντονα, δοκοί 6.4 pt, σημειώσεις 6.0 pt.
- Εκτέλεση: `python3 enlarge.py <A2_1-50.pdf> <A3_1-75.pdf> 1.2` (χρειάζεται pymupdf, shapely, γραμματοσειρές Liberation Sans).

## enlarge.py
```python
"""enlarge.py SRC_A2.pdf OUT_A3.pdf [E]
Re-lays an A2 1:50 sheet (made by sheet_s2.py) onto an A3 sheet at 1:75 (exact 2/3), keeping the original
line weights and enlarging every annotation text by E (default 1.2) wherever that does not create a new clash
(text/text, text over walls/beams/columns, dimension lines, ticks, extension lines, blue planted elements).
Texts are re-anchored by role: dimension values keep the edge facing their dimension line, beam labels and
slab circles keep their centre, labels/levels/notes keep their baseline start."""
import pymupdf, math, re, sys, collections
from shapely.geometry import LineString, box, Polygon, Point
from shapely.strtree import STRtree

MM = 72 / 25.4
SRC, OUT = sys.argv[1], sys.argv[2]
E_TARGET = float(sys.argv[3]) if len(sys.argv) > 3 else 1.2
SC = 50.0 / 75.0
FR = pymupdf.Font(fontfile="/usr/share/fonts/truetype/liberation/LiberationSans-Regular.ttf")
FB = pymupdf.Font(fontfile="/usr/share/fonts/truetype/liberation/LiberationSans-Bold.ttf")

src = pymupdf.open(SRC); P = src[0]
dr = P.get_drawings(); tt = P.get_texttrace()

# ---------------------------------------------------------------- page placement (centre the content)
bx0 = by0 = 1e9; bx1 = by1 = -1e9
for d in dr:
    r = d["rect"]; bx0, by0, bx1, by1 = min(bx0, r.x0), min(by0, r.y0), max(bx1, r.x1), max(by1, r.y1)
for t in tt:
    r = t["bbox"]; bx0, by0, bx1, by1 = min(bx0, r[0]), min(by0, r[1]), max(bx1, r[2]), max(by1, r[3])
W3, H3 = 420 * MM, 297 * MM
OFFX = (W3 - (bx1 - bx0) * SC) / 2 - bx0 * SC
OFFY = (H3 - (by1 - by0) * SC) / 2 - by0 * SC
def T(x, y): return (x * SC + OFFX, y * SC + OFFY)

# ---------------------------------------------------------------- texts and roles
TB_X, TB_Y = 520 * MM, 355 * MM        # title box area on the A2 sheet
dimlines = []                          # 0.35 black axis-aligned lines (dimension lines)
for d in dr:
    if d["type"] == "s" and round(d.get("width") or 0, 2) == 0.35 and (d.get("color") or (1,))[0] < 0.1:
        for it in d["items"]:
            if it[0] == "l":
                a, b = it[1], it[2]
                if abs(a.y - b.y) < 0.3 or abs(a.x - b.x) < 0.3: dimlines.append((a.x, a.y, b.x, b.y))
circles = []                           # slab circles (r ≈ 2.8 mm) and level circles (r ≈ 1.3 mm)
for d in dr:
    if d["type"] == "fs" and round(d.get("width") or 0, 2) == 0.35 and all(it[0] == "c" for it in d["items"]):
        r = d["rect"]; circles.append(dict(cx=(r.x0 + r.x1) / 2, cy=(r.y0 + r.y1) / 2, r=r.width / 2))
slab_circles = [c for c in circles if c["r"] > 6]
masks = [d["rect"] for d in dr if d["type"] == "f" and d.get("fill") and min(d["fill"]) > 0.99]

NUDGE = {"Κ30  50/30": (0.0, 37.0),          # A2 pt: K30 label below the south wall (like K28), clear of the ramp outline
         "Κ25  25/50": (-52.0, 20.0),         # east K25 label clear of the east-strip chain V3
         "1.60": (0.0, 15.0),                 # core inner height: value moved off the 1.80 line
         "Φελιζόλ": (0.0, -14.0),             # note moved off the 2.15 dimension line
         "Δ8.1  30/50  L=5.71": (-12.0, 0.0)} # beam label clear of the 0.20 dimension tick
FIX = {"Δ2.1": (-5.9, 1.1), "40/50": (-4.7, 1.1), "L=1.35": (-3.5, 1.1)}   # Δ2.1 3-line label between the blue level symbol and chain H1
CF = 1.25                                    # slab circles grow 25 % more than their texts so "h=20" fits inside
texts = []
for t in tt:
    s = "".join(chr(c[0]) for c in t["chars"])
    o = t["chars"][0][2]; bb = t["bbox"]
    vert = abs(t["dir"][1]) > 0.5
    cx, cy = (bb[0] + bb[2]) / 2, (bb[1] + bb[3]) / 2
    if bb[0] >= TB_X and bb[1] >= TB_Y: role = "title"
    elif re.match(r"^\d+\.\d\d$", s): role = "dim"
    elif re.match(r"^(Π\d*|h=\d+)$", s): role = "slab"
    elif s.startswith("Δ") and "L=" in s: role = "beam"
    else: role = "label"
    if s in NUDGE or s in FIX:
        dx, dy = NUDGE[s] if s in NUDGE else (0.0, FIX[s][0])
        o = (o[0] + dx, o[1] + dy); bb = (bb[0] + dx, bb[1] + dy, bb[2] + dx, bb[3] + dy); cx += dx; cy += dy
    rec = dict(s=s, o=(o[0], o[1]), bb=tuple(bb), vert=vert, bold="Bold" in t["font"], size=t["size"], color=tuple(t["color"]), role=role)
    # anchor
    if role == "dim":
        best = None
        for lx0, ly0, lx1, ly1 in dimlines:
            if abs(ly0 - ly1) < 0.3:     # horizontal dimension line
                if min(lx0, lx1) - 3 <= cx <= max(lx0, lx1) + 3:
                    dist = 0 if bb[1] <= ly0 <= bb[3] else min(abs(ly0 - bb[1]), abs(ly0 - bb[3]))
                    if best is None or dist < best[0]: best = (dist, "H", ly0)
            else:                        # vertical dimension line
                if min(ly0, ly1) - 3 <= cy <= max(ly0, ly1) + 3:
                    dist = 0 if bb[0] <= lx0 <= bb[2] else min(abs(lx0 - bb[0]), abs(lx0 - bb[2]))
                    if best is None or dist < best[0]: best = (dist, "V", lx0)
        if best and best[0] < 15:
            if best[1] == "H": rec["A"] = (cx, bb[3]) if cy < best[2] else (cx, bb[1])
            else: rec["A"] = (bb[2], cy) if cx < best[2] else (bb[0], cy)
        else: rec["A"] = (cx, cy)
    elif role == "slab":
        c = min(slab_circles, key=lambda c: math.hypot(c["cx"] - cx, c["cy"] - cy))
        rec["A"] = (c["cx"], c["cy"]); rec["circle"] = slab_circles.index(c)
    elif role == "beam": rec["A"] = (cx, cy)
    else: rec["A"] = rec["o"]
    rec["masked"] = any(m.contains(pymupdf.Point(cx, cy)) for m in masks)
    texts.append(rec)

# multi-line labels/notes: lines stacked at the same x keep their proportions (common anchor = first line's origin)
lab = [i for i, r in enumerate(texts) if r["role"] == "label" and not r["vert"] and r["s"] not in FIX]
lab.sort(key=lambda i: (round(texts[i]["o"][0]), texts[i]["o"][1]))
for a, b in zip(lab, lab[1:]):
    ra, rb = texts[a], texts[b]
    if abs(ra["o"][0] - rb["o"][0]) < 0.6 and abs(ra["size"] - rb["size"]) < 0.01 and 0 < rb["o"][1] - ra["o"][1] < 1.6 * ra["size"]:
        rb["A"] = ra["A"]; rb["block"] = ra.get("block", a); ra["block"] = ra.get("block", a)

def placed(rec, E):
    g = SC * E; Ax, Ay = rec["A"]; Apx, Apy = T(Ax, Ay)
    o = (Apx + (rec["o"][0] - Ax) * g, Apy + (rec["o"][1] - Ay) * g)
    b = (Apx + (rec["bb"][0] - Ax) * g, Apy + (rec["bb"][1] - Ay) * g, Apx + (rec["bb"][2] - Ax) * g, Apy + (rec["bb"][3] - Ay) * g)
    return o, b
def tpoly(b, f=0.10):
    h = min(b[2] - b[0], b[3] - b[1]) * f
    return box(b[0] + h, b[1] + h, b[2] - h, b[3] - h)

# ---------------------------------------------------------------- obstacles (A3 coordinates)
obst = []
for d in dr:
    if d["type"] != "s": continue
    w = round(d.get("width") or 0, 2)
    if w < 0.25: continue
    for it in d["items"]:
        if it[0] == "l":
            a, b = it[1], it[2]
            axis = abs(a.x - b.x) < 0.3 or abs(a.y - b.y) < 0.3
            tick = w == 0.6 and abs(abs(a.x - b.x) - abs(a.y - b.y)) < 0.3 and math.hypot(b.x - a.x, b.y - a.y) < 8
            if w <= 0.3 and not axis: continue          # thin diagonals (X crosses, leaders) – crossing them is acceptable
            if not axis and w >= 0.6 and not tick and (d.get("color") or (0,))[2] > 0.5: continue
            obst.append(LineString([T(a.x, a.y), T(b.x, b.y)]))
        elif it[0] == "re":
            r = it[1]; obst.append(box(*T(r.x0, r.y0), *T(r.x1, r.y1)).exterior)
        elif it[0] == "c":
            obst.append(LineString([T(p.x, p.y) for p in it[1:5]]))
small_circles = [Point(*T(c["cx"], c["cy"])).buffer(c["r"] * SC) for c in circles if c["r"] <= 6]
obst += small_circles
tree = STRtree(obst)

def hits(poly):
    return set(int(i) for i in tree.query(poly) if obst[int(i)].intersects(poly))

# ---------------------------------------------------------------- baseline (pure 2/3 scaling) and greedy enlargement
target = {"title": 1.0, "dim": E_TARGET, "slab": E_TARGET, "beam": E_TARGET, "label": E_TARGET}
E = [FIX[r["s"]][1] if r["s"] in FIX else target[r["role"]] for r in texts]
PINNED = set(i for i, r in enumerate(texts) if r["s"] in FIX)
def geoms(Evec):
    out = []
    for i, r in enumerate(texts):
        o, b = placed(r, Evec[i])
        out.append((o, b, box(b[0] - 0.3, b[1] - 0.3, b[2] + 0.3, b[3] + 0.3) if r["masked"] else tpoly(b)))
    return out
def circle_polys(Evec):
    cp = {}
    for i, r in enumerate(texts):
        if r["role"] == "slab": cp[r["circle"]] = max(cp.get(r["circle"], 1.0), Evec[i])
    return {k: Point(*T(slab_circles[k]["cx"], slab_circles[k]["cy"])).buffer(slab_circles[k]["r"] * SC * (e if e == 1.0 else e * CF)) for k, e in cp.items()}
base = geoms([1.0] * len(texts)); base_c = circle_polys([1.0] * len(texts))
base_hits = [hits(g[2]) for g in base]
base_circ = [set(k for k, c in base_c.items() if texts[i]["role"] != "slab" and c.intersects(base[i][2])) for i in range(len(texts))]
def pairs(gs):
    tr = STRtree([g[2] for g in gs]); out = set()
    for i, g in enumerate(gs):
        for j in tr.query(g[2]):
            j = int(j)
            if j > i and gs[j][2].intersects(g[2]):
                # the two lines of a slab label and the lines of a multi-line note are allowed to touch their partner
                out.add((i, j))
    return out
base_pairs = pairs(base)
print("baseline: text pairs touching", len(base_pairs))
for it in range(40):
    gs = geoms(E); cps = circle_polys(E)
    bad = set()
    for i, g in enumerate(gs):
        if E[i] <= 1.0: continue
        if hits(g[2]) - base_hits[i]: bad.add(i)
        if texts[i]["role"] != "slab" and set(k for k, c in cps.items() if c.intersects(g[2])) - base_circ[i]: bad.add(i)
    for i, j in pairs(gs) - base_pairs:
        if E[i] > 1.0: bad.add(i)
        if E[j] > 1.0: bad.add(j)
    # a slab circle that grew onto an obstacle shrinks its two texts
    for k, c in cps.items():
        if hits(c.exterior) - hits(base_c[k].exterior):
            for i, r in enumerate(texts):
                if r["role"] == "slab" and r["circle"] == k and E[i] > 1.0: bad.add(i)
    if not bad: break
    for i in list(bad):
        if "block" in texts[i]: bad |= {j for j, r in enumerate(texts) if r.get("block") == texts[i]["block"]}
    for i in bad - PINNED: E[i] = round(max(1.0, E[i] - 0.05), 3)
    if not (bad - PINNED): break
print("iterations:", it + 1)
stat = collections.Counter((texts[i]["role"], E[i]) for i in range(len(texts)))
for k in sorted(stat): print("  ", k, stat[k])
for i, r in enumerate(texts):
    if r["role"] != "title" and E[i] < target[r["role"]]: print("   reduced:", r["s"], E[i])

# ---------------------------------------------------------------- audit (final layout): texts over dimension/extension lines or ticks
aud = []
for d in dr:
    if d["type"] != "s": continue
    w = round(d.get("width") or 0, 2)
    for it in d["items"]:
        if it[0] != "l": continue
        a, b = it[1], it[2]
        tick = w == 0.6 and abs(abs(a.x - b.x) - abs(a.y - b.y)) < 0.3 and math.hypot(b.x - a.x, b.y - a.y) < 8
        if w in (0.35, 0.25) or tick: aud.append((LineString([T(a.x, a.y), T(b.x, b.y)]), "dim" if w == 0.35 else ("ext" if w == 0.25 else "tick")))
gsf = geoms(E)
def glyph_box(i):
    r = texts[i]; o = gsf[i][0]; size = r["size"] * SC * E[i]; b = gsf[i][1]
    if not r["vert"]: return (b[0] - 0.4, o[1] - 0.74 * size - 0.3, b[2] + 0.4, o[1] + 0.06 * size)
    return (o[0] - 0.74 * size - 0.3, b[1] - 0.4, o[0] + 0.06 * size, b[3] + 0.4)
for i, g in enumerate(gsf):
    if texts[i]["role"] == "title": continue
    k = [kind for ln, kind in aud if ln.intersects(tpoly(g[1], 0.15))]
    if k and all(x == "ext" for x in k) and texts[i]["role"] == "dim" and not texts[i]["masked"]:
        texts[i]["gmask"] = glyph_box(i); continue
    if k: print("   AUDIT text over", sorted(set(k)), "|", texts[i]["s"], "at mm (%.0f,%.0f)" % (g[1][0] / MM, g[1][1] / MM))
print("glyph masks for values crossed by extension lines:", sum(1 for r in texts if "gmask" in r))

# ---------------------------------------------------------------- draw
out = pymupdf.open(); Q = out.new_page(width=W3, height=H3)
sh = Q.new_shape()
for d in dr:
    for it in d["items"]:
        if it[0] == "l": sh.draw_line(pymupdf.Point(*T(it[1].x, it[1].y)), pymupdf.Point(*T(it[2].x, it[2].y)))
        elif it[0] == "c": sh.draw_bezier(*[pymupdf.Point(*T(p.x, p.y)) for p in it[1:5]])
        elif it[0] == "re": r = it[1]; sh.draw_rect(pymupdf.Rect(*T(r.x0, r.y0), *T(r.x1, r.y1)))
        elif it[0] == "qu": q = it[1]; sh.draw_quad(pymupdf.Quad(*[pymupdf.Point(*T(p.x, p.y)) for p in (q.ul, q.ur, q.ll, q.lr)]))
    lc = d.get("lineCap"); lc = lc[0] if isinstance(lc, (tuple, list)) else (lc or 0)
    sh.finish(width=d.get("width") or 0, color=d.get("color"), fill=d.get("fill"), closePath=d.get("closePath", False),
              even_odd=d.get("even_odd", False), lineCap=lc, lineJoin=int(d.get("lineJoin") or 0),
              fill_opacity=d.get("fill_opacity") if d.get("fill_opacity") is not None else 1,
              stroke_opacity=d.get("stroke_opacity") if d.get("stroke_opacity") is not None else 1)
sh.commit()
gs = geoms(E); cps = circle_polys(E)
sh = Q.new_shape()
for k, c in cps.items():      # enlarged slab circles (cover the old ones)
    sc = slab_circles[k]; ctr = pymupdf.Point(*T(sc["cx"], sc["cy"])); e = max(E[i] for i, r in enumerate(texts) if r["role"] == "slab" and r["circle"] == k)
    rr = sc["r"] * SC * (e if e == 1.0 else e * CF)
    sh.draw_circle(ctr, rr); sh.finish(width=0.35, color=(0, 0, 0), fill=(1, 1, 1))
    sh.draw_line(pymupdf.Point(ctr.x - rr, ctr.y), pymupdf.Point(ctr.x + rr, ctr.y)); sh.finish(width=0.35, color=(0, 0, 0))
for i, r in enumerate(texts):  # enlarged masks
    if r["masked"]:
        b = gs[i][1]; sh.draw_rect(pymupdf.Rect(b[0] - 0.3, b[1] - 0.3, b[2] + 0.3, b[3] + 0.3)); sh.finish(width=0, color=None, fill=(1, 1, 1))
    elif "gmask" in r:
        sh.draw_rect(pymupdf.Rect(*r["gmask"])); sh.finish(width=0, color=None, fill=(1, 1, 1))
sh.commit()
for i, r in enumerate(texts):
    s = r["s"]
    if s == "ΚΛΙΜΑΚΑ 1:50 (Α2)": s = "ΚΛΙΜΑΚΑ 1:75 (Α3)"
    o = gs[i][0]; size = r["size"] * SC * E[i]
    tw = pymupdf.TextWriter(Q.rect)
    tw.append(pymupdf.Point(*o), s, font=FB if r["bold"] else FR, fontsize=size)
    if r["vert"]: tw.write_text(Q, color=r["color"], morph=(pymupdf.Point(*o), pymupdf.Matrix(1, 0, 0, 1, 0, 0).prerotate(90)))
    else: tw.write_text(Q, color=r["color"])
meta = src.metadata or {}
out.set_metadata({"title": (meta.get("title") or "Σ2").replace("1:50", "1:75") + " – A3 1:75"})
out.save(OUT, garbage=4, deflate=True)
print("margins mm: x %.1f / %.1f  y %.1f / %.1f" % ((bx0 * SC + OFFX) / MM, (W3 - (bx1 * SC + OFFX)) / MM, (by0 * SC + OFFY) / MM, (H3 - (by1 * SC + OFFY)) / MM))
print("saved", OUT)
```
